def evaluate(name, y_true, y_pred, y_prob=None):
    acc  = accuracy_score(y_true, y_pred)
    f1   = f1_score(y_true, y_pred, average="weighted")
    prec = precision_score(y_true, y_pred, average="weighted", zero_division=0)
    rec  = recall_score(y_true, y_pred, average="weighted", zero_division=0)
    auc  = roc_auc_score(y_true, y_prob) if y_prob is not None else None
    cm   = confusion_matrix(y_true, y_pred)
    print(f"\n{'─'*60}\n  ✅ {name}\n{'─'*60}")
    print(f"  Accuracy  : {acc*100:.2f}%  |  Precision: {prec*100:.2f}%")
    print(f"  Recall    : {rec*100:.2f}%  |  F1 Score : {f1*100:.2f}%")
    if auc: print(f"  AUC-ROC   : {auc:.4f}")
    print(f"  Confusion Matrix:\n{cm}")

    # CloudWatch-compatible metric logging
    print(f"metric:accuracy={acc*100:.4f};")
    if auc: print(f"metric:auc={auc:.4f};")

    RESULTS[name] = dict(accuracy=round(acc*100,2), f1=round(f1*100,2),
                         auc=round(auc,4) if auc else None,
                         confusion_matrix=cm.tolist())

def save_plot(fig, name):
    path = f"{SM_OUTPUT_DIR}/{name}.png"
    fig.savefig(path, dpi=150, bbox_inches="tight")
    plt.close(fig)
    print(f"  📊 Saved plot → {path}")

def kfold_cv(model, X, y, k=5):
    skf    = StratifiedKFold(n_splits=k, shuffle=True, random_state=42)
    scores = cross_val_score(model, X, y, cv=skf, scoring="roc_auc", n_jobs=-1)
    print(f"  📐 {k}-Fold CV AUC: {scores.mean():.4f} ± {scores.std():.4f}")
    return scores.mean(), scores.std()

def save_model(obj, filename):
    path = f"{SM_MODEL_DIR}/{filename}"
    joblib.dump(obj, path)
    print(f"  💾 Saved: {filename} → {path}")
    return path

def xgb_obj(trial):
    p = dict(
        n_estimators    = trial.suggest_int("n_estimators", 100, 500),
        max_depth       = trial.suggest_int("max_depth", 3, 10),
        learning_rate   = trial.suggest_float("learning_rate", 0.01, 0.3, log=True),
        subsample       = trial.suggest_float("subsample", 0.6, 1.0),
        colsample_bytree= trial.suggest_float("colsample_bytree", 0.6, 1.0),
        use_label_encoder=False, eval_metric="auc",
        tree_method="gpu_hist" if DEVICE == "cuda" else "hist",
    )
    m = XGBClassifier(**p, random_state=42)
    m.fit(Xtr1, ytr1, eval_set=[(Xvl1, yvl1)], verbose=False)
    return roc_auc_score(yvl1, m.predict_proba(Xvl1)[:,1])

study1 = optuna.create_study(direction="maximize")
study1.optimize(xgb_obj, n_trials=30)

m1 = XGBClassifier(**study1.best_params, use_label_encoder=False, eval_metric="auc",
                   tree_method="gpu_hist" if DEVICE=="cuda" else "hist", random_state=42)

class BiLSTM(nn.Module):
    def __init__(self, inp, h=128, layers=2):
        super().__init__()
        self.lstm = nn.LSTM(inp, h, layers, batch_first=True, dropout=0.3, bidirectional=True)
        self.fc   = nn.Sequential(nn.Linear(h*2,64), nn.ReLU(), nn.Dropout(0.3),
                                   nn.Linear(64,1), nn.Sigmoid())
    def forward(self, x):
        out, _ = self.lstm(x)
        return self.fc(out[:,-1,:]).squeeze(1)

Xt2 = torch.FloatTensor(Xtr2).unsqueeze(1).to(DEVICE)
yt2 = torch.FloatTensor(ytr2).to(DEVICE)
Xv2 = torch.FloatTensor(Xte2).unsqueeze(1).to(DEVICE)

lstm2 = BiLSTM(Xtr2.shape[1]).to(DEVICE)

crit2 = nn.BCELoss()
tl2, vl2 = [], []
yv2t = torch.FloatTensor(yte2).to(DEVICE)

print("  Training Bi-LSTM — 50 epochs")
for ep in range(50):
    lstm2.train(); ep_l=0
    for Xb, yb in dl2:
        opt2.zero_grad(); loss=crit2(lstm2(Xb),yb)
        loss.backward(); nn.utils.clip_grad_norm_(lstm2.parameters(),1.0)
        opt2.step(); ep_l += loss.item()
    lstm2.eval()
    with torch.no_grad(): vl=crit2(lstm2(Xv2), yv2t).item()
    tl2.append(ep_l/len(dl2)); vl2.append(vl); sch2.step()
    if (ep+1)%10==0: print(f"  Ep {ep+1}/50  Train:{tl2[-1]:.4f}  Val:{vl:.4f}")

lstm2.eval()

class TwinNet(nn.Module):
    def __init__(self, inp):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(inp,256), nn.BatchNorm1d(256), nn.ReLU(), nn.Dropout(.25),
            nn.Linear(256,128), nn.BatchNorm1d(128), nn.ReLU(), nn.Dropout(.2),
            nn.Linear(128,64),  nn.ReLU(),
            nn.Linear(64,1),    nn.Sigmoid(),
        )
    def forward(self,x): return self.net(x).squeeze(1)

Xt5 = torch.FloatTensor(Xtr5).to(DEVICE); yt5=torch.FloatTensor(ytr5.values).to(DEVICE)
Xv5 = torch.FloatTensor(Xte5).to(DEVICE); yv5=torch.FloatTensor(yte5.values).to(DEVICE)
dl5 = DataLoader(TensorDataset(Xt5,yt5), batch_size=512, shuffle=True)
twin = TwinNet(Xtr5.shape[1]).to(DEVICE)
opt5 = optim.Adam(twin.parameters(), lr=1e-3)
sch5 = optim.lr_scheduler.ReduceLROnPlateau(opt5, patience=5, factor=0.5)
mse5 = nn.MSELoss(); tl5,vl5=[],[]

print("  Training Digital Twin MLP — 60 epochs")
for ep in range(60):
    twin.train(); ep_l=0
    for Xb,yb in dl5:
        opt5.zero_grad(); loss=mse5(twin(Xb),yb); loss.backward(); opt5.step(); ep_l+=loss.item()
    twin.eval()
    with torch.no_grad(): vl=mse5(twin(Xv5),yv5).item()
    tl5.append(ep_l/len(dl5)); vl5.append(vl); sch5.step(vl)
    if (ep+1)%15==0: print(f"  Ep {ep+1}/60  Train MSE:{tl5[-1]:.5f}  Val MSE:{vl:.5f}")

twin.eval()
with torch.no_grad(): preds5=twin(Xv5).cpu().numpy()
mse_v=mean_squared_error(yte5,preds5); r2_v=r2_score(yte5,preds5)
print(f"\n  ✅ Digital Twin — MSE:{mse_v:.5f}  R²:{r2_v:.4f}  RMSE:{np.sqrt(mse_v):.5f}")
print(f"metric:r2={r2_v:.4f};")
RESULTS["Digital Twin — NHANES"] = dict(mse=round(mse_v,5), r2=round(r2_v,4))

torch.save(twin.state_dict(), f"{SM_MODEL_DIR}/digital_twin.pth")
save_model({"scaler": sc5, "feature_names": list(df5.columns)}, "digital_twin_meta.pkl")

#  MODEL 6: FEDERATED LEARNING — FedAvg 3 Nodes
# ════════════════════════════════════════════════════════════════════════════
print("\n\n🔬 MODEL 6: Federated Learning — FedAvg (3 Hospital Nodes)")

X6, y6 = make_classification(n_samples=15000, n_features=20, n_informative=12,
                              n_redundant=4, weights=[.6,.4], random_state=42)
sc6 = StandardScaler(); X6=sc6.fit_transform(X6)
nodes_X = [X6[:5000], X6[5000:10000], X6[10000:]]
nodes_y = [y6[:5000], y6[5000:10000], y6[10000:]]
node_names = ["BIDMC (Boston)","Mayo Clinic","Johns Hopkins"]
fed_log=[]; global_w=None

def train_node(Xn, yn):
    Xtr_n, Xte_n, ytr_n, yte_n = train_test_split(Xn, yn, test_size=0.2, random_state=42)
    m = LGBMClassifier(n_estimators=100, learning_rate=0.1, random_state=42, verbose=-1)
    m.fit(Xtr_n, ytr_n)
    auc = roc_auc_score(yte_n, m.predict_proba(Xte_n)[:,1])
    w   = m.feature_importances_.astype(float); return auc, w/w.sum()

print("  FedAvg — 10 rounds across 3 simulated hospital nodes")
for rnd in range(1, 11):
    aucs, weights = [], []
    for Xn, yn in zip(nodes_X, nodes_y):
        auc, w = train_node(Xn, yn); aucs.append(auc); weights.append(w)
    global_w = np.average(weights, axis=0, weights=[1/3]*3)
    avg_auc  = np.mean(aucs)
    fed_log.append({"round": rnd, "global_auc": round(avg_auc,4),
                    "node_aucs": [round(a,4) for a in aucs]})
    print(f"  Round {rnd:2d} | Global AUC:{avg_auc:.4f} | Nodes:{[f'{a:.4f}' for a in aucs]}")
    print(f"metric:federated_auc={avg_auc:.4f};")

RESULTS["Federated Learning"] = {"global_auc": fed_log[-1]["global_auc"]}
fig, ax = plt.subplots(figsize=(9,4))
ax.plot([m["round"] for m in fed_log],[m["global_auc"] for m in fed_log],
        marker="o", color="#3b82f6", linewidth=2)
ax.set_title("Federated Learning — Convergence", fontweight="bold")
ax.set_xlabel("Round"); ax.set_ylabel("Global AUC")
save_plot(fig, "federated_convergence")
save_model({"global_weights": global_w, "rounds": fed_log, "scaler": sc6, "node_names": node_names},
           "federated_model.pkl")
